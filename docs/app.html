 99def extract_predictions(response: Any) -> list[dict[str, Any]]:
100    """Extract a list of predictions from common workflow response shapes.
101
102    Args:
103        response: Raw object returned by ``client.run_workflow``.
104
105    Returns:
106        A list of prediction dictionaries. An empty list means no predictions
107        were returned; it does not necessarily prove that an image is defect-free.
108
109    Raises:
110        ValueError: If the response does not contain a recognised prediction list.
111    """
112    # Workflow responses are often a list containing a dictionary. Accept
113    # either that shape or a dictionary directly to make parsing less brittle.
114    payload = response[0] if isinstance(response, list) and response else response
115
116    if not isinstance(payload, dict):
117        raise ValueError("Unexpected workflow response: expected a dictionary.")
118
119    predictions_obj = payload.get("predictions")
120
121    # Common object-detection response:
122    # {"predictions": {"predictions": [...]}}
123    if isinstance(predictions_obj, dict):
124        predictions = predictions_obj.get("predictions")
125        if isinstance(predictions, list):
126            return predictions
127
128    # Also accept a direct list:
129    # {"predictions": [...]}
130    if isinstance(predictions_obj, list):
131        return predictions_obj
132
133    # Some workflows use a named output. Rather than silently treating an
134    # unknown response as "no defects", raise a clear error for inspection.
135    raise ValueError(
136        "Could not find a prediction list in the workflow response. "
137        "Check the workflow's output format."
138    )

141def validate_prediction(prediction: dict[str, Any]) -> dict[str, Any] | None:
142    """Validate and normalise one centre-based bounding-box prediction.
143
144    Expected coordinates are x/y centre, width, and height, with confidence
145    between 0 and 1. Invalid entries are ignored rather than crashing the batch.
146
147    Args:
148        prediction: Raw prediction from the workflow.
149
150    Returns:
151        A normalised dictionary, or None if required fields are invalid.
152    """
153    try:
154        label = str(prediction["class"])
155        confidence = float(prediction["confidence"])
156        x = float(prediction["x"])
157        y = float(prediction["y"])
158        width = float(prediction["width"])
159        height = float(prediction["height"])
160    except (KeyError, TypeError, ValueError):
161        return None
162
163    if not label or not all(
164        pd.notna(value) for value in (confidence, x, y, width, height)
165    ):
166        return None
167    if not (0.0 <= confidence <= 1.0) or width <= 0 or height <= 0:
168        return None
169
170    return {
171        "class": label,
172        "confidence": confidence,
173        "x_center": x,
174        "y_center": y,
175        "box_width": width,
176        "box_height": height,
177    }

180def annotate_image(
181    image: Image.Image,
182    predictions: list[dict[str, Any]],
183    threshold: float,
184) -> tuple[Image.Image, list[dict[str, Any]]]:
185    """Draw qualifying bounding boxes and labels on a copy of an image.
186
187    Args:
188        image: Original image.
189        predictions: Normalised predictions.
190        threshold: Minimum confidence required for a prediction to be shown.
191
192    Returns:
193        A tuple containing the annotated image and qualifying predictions.
194
195    Note:
196        This assumes the workflow's coordinates refer to the original uploaded
197        image dimensions. Confirm this visually for your specific workflow.
198    """
199    annotated = image.convert("RGB").copy()
200    draw = ImageDraw.Draw(annotated)
201    image_width, image_height = annotated.size
202    accepted: list[dict[str, Any]] = []
203
204    try:
205        font = ImageFont.load_default()
206    except Exception:
207        font = None
208
209    for prediction in predictions:
210        if prediction["confidence"] < threshold:
211            continue
212
213        # Convert centre-based coordinates to corner coordinates.
214        x = prediction["x_center"]
215        y = prediction["y_center"]
216        width = prediction["box_width"]
217        height = prediction["box_height"]
218
219        x1 = max(0, min(x - width / 2, image_width - 1))
220        y1 = max(0, min(y - height / 2, image_height - 1))
221        x2 = max(0, min(x + width / 2, image_width - 1))
222        y2 = max(0, min(y + height / 2, image_height - 1))
223
224        # Ignore boxes that collapse after clipping.
225        if x2 <= x1 or y2 <= y1:
226            continue
227
228        label_text = f'{prediction["class"]}: {prediction["confidence"]:.1%}'
229        draw.rectangle((x1, y1, x2, y2), outline="#ff3030", width=3)
230
231        # Add a filled label background so the text remains readable.
232        text_box = draw.textbbox((x1, max(0, y1 - 18)), label_text, font=font)
233        draw.rectangle(text_box, fill="#ff3030")
234        draw.text(
235            (x1, max(0, y1 - 18)),
236            label_text,
237            fill="white",
238            font=font,
239        )
240
241        accepted.append(prediction)
242
243    return annotated, accepted

253def run_inference(
254    client: InferenceHTTPClient,
255    uploaded_file: Any,
256) -> tuple[list[dict[str, Any]], float]:
257    """Run the hosted workflow for one uploaded image.
258
259    Args:
260        client: Configured Roboflow inference client.
261        uploaded_file: Streamlit UploadedFile object.
262
263    Returns:
264        Normalised valid predictions and elapsed inference time in seconds.
265
266    Raises:
267        ValueError: If the image or workflow response is invalid.
268        Exception: Network/API errors are passed to the caller for display.
269    """
270    suffix = Path(uploaded_file.name).suffix.lower()
271    if suffix not in SUPPORTED_EXTENSIONS:
272        raise ValueError(f"Unsupported file type: {suffix or 'no extension'}")
273
274    # The SDK accepts an image path. Use a temporary file and remove it after
275    # inference, rather than requiring users to manually save uploads.
276    uploaded_bytes = uploaded_file.getvalue()
277    if not uploaded_bytes:
278        raise ValueError("The uploaded file is empty.")
279
280    try:
281        with Image.open(io.BytesIO(uploaded_bytes)) as check_image:
282            check_image.verify()
283    except (UnidentifiedImageError, OSError) as error:
284        raise ValueError("This file is not a readable image.") from error
285
286    start = time.perf_counter()
287    with tempfile.NamedTemporaryFile(suffix=suffix) as temporary_file:
288        temporary_file.write(uploaded_bytes)
289        temporary_file.flush()
290
291        response = client.run_workflow(
292            workspace_name=WORKSPACE,
293            workflow_id=WORKFLOW_ID,
294            images={"image": temporary_file.name},
295            use_cache=False,
296        )
297
298    elapsed = time.perf_counter() - start
299    raw_predictions = extract_predictions(response)
300
301    normalised = []
302    for raw_prediction in raw_predictions:
303        if isinstance(raw_prediction, dict):
304            prediction = validate_prediction(raw_prediction)
305            if prediction is not None:
306                normalised.append(prediction)
307
308    return normalised, elapsed

311def make_zip(annotated_images: dict[str, bytes]) -> bytes:
312    """Package annotated PNG images into an in-memory ZIP archive."""
313    buffer = io.BytesIO()
314    with zipfile.ZipFile(buffer, mode="w", compression=zipfile.ZIP_DEFLATED) as archive:
315        for filename, content in annotated_images.items():
316            archive.writestr(filename, content)
317    return buffer.getvalue()

def annotate_image( image: PIL.Image.Image, predictions: list[dict[str, typing.Any]], threshold: float) -> tuple[PIL.Image.Image, list[dict[str, typing.Any]]]: 
181def annotate_image(
182    image: Image.Image,
183    predictions: list[dict[str, Any]],
184    threshold: float,
185) -> tuple[Image.Image, list[dict[str, Any]]]:
186    """Draw qualifying bounding boxes and labels on a copy of an image.
187
188    Args:
189        image: Original image.
190        predictions: Normalised predictions.
191        threshold: Minimum confidence required for a prediction to be shown.
192
193    Returns:
194        A tuple containing the annotated image and qualifying predictions.
195
196    Note:
197        This assumes the workflow's coordinates refer to the original uploaded
198        image dimensions. Confirm this visually for your specific workflow.
199    """
200    annotated = image.convert("RGB").copy()
201    draw = ImageDraw.Draw(annotated)
202    image_width, image_height = annotated.size
203    accepted: list[dict[str, Any]] = []
204
205    try:
206        font = ImageFont.load_default()
207    except Exception:
208        font = None
209
210    for prediction in predictions:
211        if prediction["confidence"] < threshold:
212            continue
213
214        # Convert centre-based coordinates to corner coordinates.
215        x = prediction["x_center"]
216        y = prediction["y_center"]
217        width = prediction["box_width"]
218        height = prediction["box_height"]
219
220        x1 = max(0, min(x - width / 2, image_width - 1))
221        y1 = max(0, min(y - height / 2, image_height - 1))
222        x2 = max(0, min(x + width / 2, image_width - 1))
223        y2 = max(0, min(y + height / 2, image_height - 1))
224
225        # Ignore boxes that collapse after clipping.
226        if x2 <= x1 or y2 <= y1:
227            continue
228
229        label_text = f'{prediction["class"]}: {prediction["confidence"]:.1%}'
230        draw.rectangle((x1, y1, x2, y2), outline="#ff3030", width=3)
231
232        # Add a filled label background so the text remains readable.
233        text_box = draw.textbbox((x1, max(0, y1 - 18)), label_text, font=font)
234        draw.rectangle(text_box, fill="#ff3030")
235        draw.text(
236            (x1, max(0, y1 - 18)),
237            label_text,
238            fill="white",
239            font=font,
240        )
241
242        accepted.append(prediction)
243
244    return annotated, accepted

254def run_inference(
255    client: InferenceHTTPClient,
256    uploaded_file: Any,
257) -> tuple[list[dict[str, Any]], float]:
258    """Run the hosted workflow for one uploaded image.
259
260    Args:
261        client: Configured Roboflow inference client.
262        uploaded_file: Streamlit UploadedFile object.
263
264    Returns:
265        Normalised valid predictions and elapsed inference time in seconds.
266
267    Raises:
268        ValueError: If the image or workflow response is invalid.
269        Exception: Network/API errors are passed to the caller for display.
270    """
271    suffix = Path(uploaded_file.name).suffix.lower()
272    if suffix not in SUPPORTED_EXTENSIONS:
273        raise ValueError(f"Unsupported file type: {suffix or 'no extension'}")
274
275    # The SDK accepts an image path. Use a temporary file and remove it after
276    # inference, rather than requiring users to manually save uploads.
277    uploaded_bytes = uploaded_file.getvalue()
278    if not uploaded_bytes:
279        raise ValueError("The uploaded file is empty.")
280
281    try:
282        with Image.open(io.BytesIO(uploaded_bytes)) as check_image:
283            check_image.verify()
284    except (UnidentifiedImageError, OSError) as error:
285        raise ValueError("This file is not a readable image.") from error
286
287    start = time.perf_counter()
288    with tempfile.NamedTemporaryFile(suffix=suffix) as temporary_file:
289        temporary_file.write(uploaded_bytes)
290        temporary_file.flush()
291
292        response = client.run_workflow(
293            workspace_name=WORKSPACE,
294            workflow_id=WORKFLOW_ID,
295            images={"image": temporary_file.name},
296            use_cache=False,
297        )
298
299    elapsed = time.perf_counter() - start
300    raw_predictions = extract_predictions(response)
301
302    normalised = []
303    for raw_prediction in raw_predictions:
304        if isinstance(raw_prediction, dict):
305            prediction = validate_prediction(raw_prediction)
306            if prediction is not None:
307                normalised.append(prediction)
308
309    return normalised, elapsed
